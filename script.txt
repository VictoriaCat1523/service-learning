        <script> // СКРИПТ ДЛЯ ЛОГИНА 
            // Находим форму на странице (убедись, что у твоей формы есть id="loginForm" или замени на актуальный)
            const loginForm = document.querySelector('form') || document.getElementById('loginForm');
        
            loginForm.addEventListener('submit', async (event) => {
                event.preventDefault(); // Отменяем стандартную перезагрузку страницы
        
                // Собираем данные из полей ввода (замени 'username' и 'password' на те id, которые прописаны в твоих тегах <input>)
                const usernameInput = document.getElementById('username').value;
                const passwordInput = document.getElementById('password').value;
        
                try {
                    // Шлем POST-запрос на наш Nginx прокси
                    const response = await fetch('/api/login', {
                        method: 'POST',
                        headers: {
                            'Content-Type': 'application/json'
                        },
                        body: JSON.stringify({
                            username: usernameInput,
                            password: passwordInput
                        })
                    });
        
                    const data = await response.json();
        
                    if (response.ok) {
                        alert('Успешный вход!');
                        // Если вход успешный, перенаправляем пользователя на главную страницу после авторизации
                        window.location.href = '/HTML/main_site_after_autorization.html'; 
                    } else {
                        // Если бэкенд вернул ошибку (например, неверный пароль)
                        alert('Ошибка: ' + (data.detail || 'Неверный логин или пароль'));
                    }
                } catch (error) {
                    console.error('Ошибка сети:', error);
                    alert('Не удалось связаться с сервером бэкенда.');
                }
            });
        </script>




        <script> // СКРИПТ ДЛЯ РЕГИСТРАЦИИ
            const registerForm = document.querySelector('form');
        
            registerForm.addEventListener('submit', async (event) => {
                event.preventDefault();
        
                const usernameInput = document.getElementById('username').value;
                const emailInput = document.getElementById('email').value; // если есть поле email
                const passwordInput = document.getElementById('password').value;
        
                try {
                    const response = await fetch('/api/register', {
                        method: 'POST',
                        headers: {
                            'Content-Type': 'application/json'
                        },
                        body: JSON.stringify({
                            username: usernameInput,
                            email: emailInput,
                            password: passwordInput
                        })
                    });
        
                    const data = await response.json();
        
                    if (response.ok) {
                        alert('Регистрация успешна! Теперь вы можете войти.');
                        window.location.href = '/HTML/login.html'; // отправляем на страницу логина
                    } else {
                        alert('Ошибка регистрации: ' + (data.detail || 'Что-то пошло не так'));
                    }
                } catch (error) {
                    console.error('Ошибка сети:', error);
                    alert('Не удалось связаться с сервером.');
                }
            });
        </script>
        

<script>
    // Ждем, пока вся страница полностью загрузится
    window.addEventListener('DOMContentLoaded', () => {
        // Заставляем скрипт найти форму, даже если у нее нет id (берем первую форму на странице)
        const form = document.querySelector('form');
        
        if (!form) {
            console.error("DevOps Error: Форма не найдена на странице login.html!");
            return;
        }

        form.addEventListener('submit', async (event) => {
            event.preventDefault(); // Запрещаем странице перезагружаться

            // ВНИМАНИЕ: Проверь, чтобы id ниже ('email' и 'password') 
            # совпадали с тем, что написано в тегах <input id="..."> в твоем HTML!
            const emailField = document.getElementById('email') or document.querySelector('input[type="email"]');
            const passwordField = document.getElementById('password') or document.querySelector('input[type="password"]');

            if (!emailField || !passwordField) {
                alert("Ошибка: Скрипт не смог найти поля ввода на странице. Проверь их id в HTML!");
                return;
            }

            console.log("DevOps Log: Данные собраны, отправляем в сеть...");

            try {
                const response = await fetch('/api/login', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json'
                    },
                    body: JSON.stringify({
                        username: emailField.value,
                        password: passwordField.value
                    })
                });

                const data = await response.json();

                if (response.ok) {
                    alert('Авторизация успешна!');
                    // Перенаправляем на страницу после успешного входа
                    window.location.href = '/HTML/main_site_after_autorization.html';
                } else {
                    alert('Ошибка: ' + (data.detail || 'Неверный логин или пароль'));
                }
            } catch (error) {
                console.error('Ошибка сети:', error);
                alert('Не удалось связаться с бэкендом.');
            }
        });
    });
</script>
